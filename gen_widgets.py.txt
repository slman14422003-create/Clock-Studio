NS='xmlns:android="http://schemas.android.com/apk/res/android"'

def tv(id=None, size=14, color='text_primary', font='sans-serif', style=None, w='wrap_content', h='wrap_content',
       extra='', kind='TextView', lines='singleLine="true"', gravity=None, mt=0, ms=0, me=0, mb=0, weight=None):
    a=[]
    if id: a.append(f'android:id="@+id/{id}"')
    a.append(f'android:layout_width="{w}"'); a.append(f'android:layout_height="{h}"')
    if weight is not None: a.append(f'android:layout_weight="{weight}"')
    for k,v in (('Top',mt),('Start',ms),('End',me),('Bottom',mb)):
        if v: a.append(f'android:layout_margin{k}="{v}dp"')
    a.append(f'android:fontFamily="{font}"'); a.append('android:includeFontPadding="false"')
    if lines: a.append('android:'+lines)
    a.append(f'android:textColor="@color/{color}"'); a.append(f'android:textSize="{size}sp"')
    if style: a.append(f'android:textStyle="{style}"')
    if gravity: a.append(f'android:gravity="{gravity}"')
    if extra: a.append(extra)
    return f'<{kind} '+' '.join(a)+' />'

def wrap(direction, body, bg='wb_adaptive_1'):
    return f'''<?xml version="1.0" encoding="utf-8"?>
<FrameLayout {NS}
    android:id="@+id/wRoot"
    android:layout_width="match_parent"
    android:layout_height="match_parent"
    android:layoutDirection="{direction}">
    <ImageView
        android:id="@+id/wBg"
        android:layout_width="match_parent"
        android:layout_height="match_parent"
        android:importantForAccessibility="no"
        android:scaleType="fitXY"
        android:src="@drawable/{bg}" />
    <ImageView
        android:id="@+id/wBorder"
        android:layout_width="match_parent"
        android:layout_height="match_parent"
        android:importantForAccessibility="no"
        android:scaleType="fitXY"
        android:src="@drawable/wb_bd_adaptive_2"
        android:visibility="gone" />
{body}
</FrameLayout>
'''

def lin(orient, inner, w='match_parent', h='match_parent', pad=16, gravity=None, weight=None, bg=None, ps=None, extra='', id=None):
    a=[]
    if id: a.append(f'android:id="@+id/{id}"')
    a+= [f'android:layout_width="{w}"',f'android:layout_height="{h}"',f'android:orientation="{orient}"']
    if weight is not None: a.append(f'android:layout_weight="{weight}"')
    if gravity: a.append(f'android:gravity="{gravity}"')
    if bg: a.append(f'android:background="@drawable/{bg}"')
    if ps: a.append(f'android:paddingStart="{ps[0]}dp" android:paddingTop="{ps[1]}dp" android:paddingEnd="{ps[2]}dp" android:paddingBottom="{ps[3]}dp"')
    elif pad: a.append(f'android:padding="{pad}dp"')
    if extra: a.append(extra)
    return '<LinearLayout '+' '.join(a)+'>\n'+inner+'\n</LinearLayout>'

def spacer(h='0dp', w=None, weight=None):
    if w is None: w = '0dp' if weight and h != '0dp' else 'match_parent'
    return f'<FrameLayout android:layout_width="{w}" android:layout_height="{h}"' + (f' android:layout_weight="{weight}"' if weight else '') + ' />'

def icon(id, res, size, tint='text_secondary'):
    return f'<ImageView android:id="@+id/{id}" android:layout_width="{size}dp" android:layout_height="{size}dp" android:importantForAccessibility="no" android:src="@drawable/{res}" android:tint="@color/{tint}" />'

def clock(id, f12, f24, size, color='text_primary', font='sans-serif-light', extra='', **k):
    k.setdefault('lines', None)
    return tv(id, size, color, font, kind='TextClock', extra=f'android:format12Hour="{f12}" android:format24Hour="{f24}" '+extra, **k)
